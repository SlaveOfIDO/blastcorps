/* hd.c's PrepareStateTransition DMAs+decompresses this fixed 0xC0-byte chunk
 * every state transition (not just for the rmonPrintf, despite the old
 * comment here - it's the actual DMA source address/size). It isn't its own
 * splat segment, but it sits at a fixed offset inside endsid5_dl's raw
 * bytes: in a fully matching build endsid5_dl_ROM_END == 0x787FD0, and
 * 0x787FD0 - 0x90 == 0x787F40 and + 0xC0 == 0x788000, exactly the old
 * hardcoded values - so anchor to endsid5_dl_ROM_END instead of the literal
 * addresses, which drift whenever anything upstream changes size under
 * NON_MATCHING and previously caused this DMA to decompress garbage. */
D_787F40 = endsid5_dl_ROM_END - 0x90;
D_788000 = D_787F40 + 0xC0;
